  \begin{scope}[xshift=-4.2cm]
    \node[font=\bfseries] at (0,2.35) {in phase: $u=x+y$};

    \node[draw=ink!20,fill=gold!10,rounded corners=4pt,inner sep=6pt]

  \end{scope}

    {$x=(u+v)/2,\qquad y=(u-v)/2$\\
     each normal combination evolves independently};
